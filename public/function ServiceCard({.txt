function ServiceCard({
  service,
  index,
  Icon,
}: {
  service: any;
  index: number;
  Icon: any;
}) {
  const isEven = index % 2 === 0;

  return (
    <div 
      className={`w-full py-16 lg:py-24 transition-colors duration-300 ${
        isEven ? "bg-white" : "bg-[#fafafa]"
      }`}
    >
      <div className="max-w-7xl mx-auto px-6 md:px-12">
        <motion.div
          initial={{ opacity: 0, y: 30 }}
          whileInView={{ opacity: 1, y: 0 }}
          viewport={{ once: true, margin: "-100px" }}
          transition={{ duration: 0.8, ease: [0.25, 1, 0.5, 1] }}
          className="grid lg:grid-cols-2 gap-12 lg:gap-20 items-center"
        >
          
          {/* TEXT CONTENT CONTAINER */}
          <div className={`space-y-6 ${isEven ? "lg:order-1" : "lg:order-2"}`}>
            
            {/* الجزء العلوي: الأيقونة + الخط الأحمر + القسم */}
            <div className="flex items-center gap-4">
              {Icon && (
                <div className="text-[#da1c2e] flex-shrink-0 opacity-80 hover:opacity-100 transition-opacity">
                  <Icon size={22} strokeWidth={1.5} />
                </div>
              )}
              <span className="w-8 h-[1.5px] bg-[#da1c2e]" />
              <span className="text-xs tracking-[0.2em] uppercase font-semibold text-[#da1c2e]">
                {service.category || "SERVICES"}
              </span>
            </div>

            {/* العنوان الرئيسي الهندسي */}
            <h2 className="text-3xl md:text-4xl lg:text-5xl font-light tracking-tight text-neutral-900">
              {service.title}
            </h2>

            {/* الوصف النصي */}
            {service.description && (
              <p className="text-neutral-500 text-sm md:text-base leading-relaxed font-light max-w-xl">
                {service.description}
              </p>
            )}

            {/* القائمة الفرعية باللون الأحمر التفاعلي الفخم */}
            <ul className="space-y-3.5 pt-4">
              {service.features?.map((feature: string, idx: number) => (
                <li key={idx} className="flex items-center gap-3 group/item cursor-pointer">
                  
                  {/* مؤشر مربع أحمر صغير حاد الزوايا يتمدد أفقياً عند التحويم */}
                  <div className="w-1.5 h-1.5 bg-[#da1c2e] transition-all duration-300 group-hover/item:w-3 flex-shrink-0" />
                  
                  {/* النص يتحول للأحمر الداكن عند التحويم ليعطي تفاعل ذكي ونظيف */}
                  <span className="text-neutral-600 group-hover/item:text-[#da1c2e] transition-colors duration-300 text-sm md:text-base font-light tracking-wide">
                    {feature}
                  </span>
                </li>
              ))}
            </ul>
          </div>

          {/* IMAGE CONTAINER */}
          <div className={`${isEven ? "lg:order-2" : "lg:order-1"}`}>
            <div className="relative w-full aspect-[16/10] md:aspect-[16/11] lg:aspect-[4/3] overflow-hidden bg-neutral-100 border border-neutral-200/60 shadow-sm">
              <motion.img
                initial={{ scale: 1.05 }}
                whileInView={{ scale: 1 }}
                transition={{ duration: 1.2, ease: [0.25, 1, 0.5, 1] }}
                src={`${API_BASE_URL}${service.image}`}
                alt={service.title}
                className="w-full h-full object-cover grayscale-[15%] hover:grayscale-0 transition-all duration-700 ease-out"
              />
            </div>
          </div>

        </motion.div>
      </div>
    </div>
  );
}
